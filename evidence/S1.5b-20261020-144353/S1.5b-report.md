### 7.1 คำสั่ง rehash อิสระ (ทำซ้ำได้ — ผลที่ได้: `changed=1` = `PlayerprefData.dat`, `missing=0`, `added=0`)
```bash
python - <<'PY'
import hashlib, os
base_dir = r"C:\Users\memph\LongYinCoop\coop-test\S1-20261020-113926"
save_dir = r"C:\Program Files (x86)\Steam\steamapps\common\LongYinLiZhiZhuan\LongYinLiZhiZhuan_Data\Save"
base = {}
for line in open(os.path.join(base_dir, "savehash.real.before.txt"), encoding="utf-8"):
    line = line.strip()
    if not line:
        continue
    h, rel = line.split(None, 1)
    rel = rel.strip().lstrip("*").replace("\\", "/")
    if rel.startswith("Save/"):
        rel = rel[5:]
    base[rel] = h
cur = {}
for root, _, files in os.walk(save_dir):
    for f in files:
        p = os.path.join(root, f)
        rel = os.path.relpath(p, save_dir).replace("\\", "/")
        cur[rel] = hashlib.sha256(open(p, "rb").read()).hexdigest()
print("baseline", len(base), "current", len(cur))
print("changed", [k for k in base if k in cur and base[k] != cur[k]])
print("missing", [k for k in base if k not in cur])
print("added", [k for k in cur if k not in base])
```
